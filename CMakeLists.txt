cmake_minimum_required(VERSION 3.20)
project(jpp VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(LLVM REQUIRED CONFIG)
if(LLVM_VERSION_MAJOR LESS 18)
  message(FATAL_ERROR "J++ needs LLVM/Clang 18 or newer")
endif()
find_package(Clang REQUIRED CONFIG HINTS "${LLVM_INSTALL_PREFIX}/lib/cmake/clang")
set(JPP_CLANG_RESOURCE_DIR "" CACHE PATH "Override Clang's builtin-header directory")
set(jpp_resource_dir "${JPP_CLANG_RESOURCE_DIR}")
if(NOT jpp_resource_dir)
  execute_process(COMMAND "${LLVM_TOOLS_BINARY_DIR}/clang" --print-resource-dir
    OUTPUT_VARIABLE jpp_resource_dir OUTPUT_STRIP_TRAILING_WHITESPACE
    RESULT_VARIABLE jpp_resource_result)
  if(NOT jpp_resource_result EQUAL 0)
    message(FATAL_ERROR "Cannot locate Clang builtin headers; set JPP_CLANG_RESOURCE_DIR")
  endif()
endif()
if(NOT EXISTS "${jpp_resource_dir}/include/stddef.h")
  message(FATAL_ERROR "JPP_CLANG_RESOURCE_DIR must contain Clang's include/stddef.h")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/runtime/Runtime.java" JPP_RUNTIME)
# The emitted source is formatted separately. Keep the embedded copy compact.
string(REGEX REPLACE "\n[ \t]+" "\n" JPP_RUNTIME "${JPP_RUNTIME}")
# Keep individual literals within the C++ implementation limits as the runtime grows.
set(JPP_RUNTIME_PARTS "")
string(LENGTH "${JPP_RUNTIME}" JPP_RUNTIME_LENGTH)
while(JPP_RUNTIME_LENGTH GREATER 0)
  if(JPP_RUNTIME_LENGTH GREATER 16384)
    set(JPP_RUNTIME_CHUNK_SIZE 16384)
  else()
    set(JPP_RUNTIME_CHUNK_SIZE ${JPP_RUNTIME_LENGTH})
  endif()
  string(SUBSTRING "${JPP_RUNTIME}" 0 ${JPP_RUNTIME_CHUNK_SIZE} JPP_RUNTIME_CHUNK)
  string(APPEND JPP_RUNTIME_PARTS "R\"JPP_RUNTIME(${JPP_RUNTIME_CHUNK})JPP_RUNTIME\",\n")
  string(SUBSTRING "${JPP_RUNTIME}" ${JPP_RUNTIME_CHUNK_SIZE} -1 JPP_RUNTIME)
  string(LENGTH "${JPP_RUNTIME}" JPP_RUNTIME_LENGTH)
endwhile()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS runtime/Runtime.java)
configure_file(src/runtime.hpp.in generated/runtime.hpp @ONLY)
add_executable(jpp src/main.cpp src/transpiler.cpp)
set_target_properties(jpp PROPERTIES OUTPUT_NAME "j++")
target_include_directories(jpp SYSTEM PRIVATE ${CLANG_INCLUDE_DIRS} ${LLVM_INCLUDE_DIRS})
target_compile_definitions(jpp PRIVATE JPP_CLANG_RESOURCE_DIR="${jpp_resource_dir}")
target_include_directories(jpp PRIVATE src "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(jpp PRIVATE clang-cpp LLVM)
target_compile_options(jpp PRIVATE -Wall -Wextra -Wpedantic)
include(GNUInstallDirs)
install(TARGETS jpp RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
enable_testing()
option(JPP_TEST_GNU_FRONTEND "Test GNU extensions requiring the compatibility frontend" OFF)
add_executable(jpp-tests tests/integration.cpp)
if(JPP_TEST_GNU_FRONTEND)
  target_compile_definitions(jpp-tests PRIVATE JPP_TEST_GNU_FRONTEND=1)
endif()
target_compile_definitions(jpp-tests PRIVATE JPP_BINARY="$<TARGET_FILE:jpp>" JPP_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME integration COMMAND jpp-tests)
